const makeId = (type: string) =>
  `${type}-${Date.now().toString(36)}-${Math.random()
    .toString(36)
    .slice(2, 8)}`;

function cloneNode(node: BuilderNode): BuilderNode {
  return {
    ...node,
    id: makeId(node.type),
    props: structuredClone(node.props || {}),
    styles: structuredClone(node.styles || {}),
    advanced: node.advanced ? structuredClone(node.advanced) : undefined,
    children: (node.children || []).map(cloneNode),
  };
}

function findNode(nodes: BuilderNode[], id: string | null): BuilderNode | null {
  if (!id) return null;

  for (const node of nodes) {
    if (node.id === id) return node;

    const found = findNode(node.children || [], id);

    if (found) return found;
  }

  return null;
}

function findParent(
  nodes: BuilderNode[],
  id: string,
  parentId: string | null = null,
): string | null | undefined {
  for (const node of nodes) {
    if (node.id === id) return parentId;

    const result = findParent(node.children || [], id, node.id);

    if (result !== undefined) return result;
  }

  return undefined;
}

function updateTree(
  nodes: BuilderNode[],
  id: string,
  updater: (node: BuilderNode) => BuilderNode,
): BuilderNode[] {
  return nodes.map((node) => {
    if (node.id === id) {
      return updater(node);
    }

    return {
      ...node,
      children: updateTree(node.children || [], id, updater),
    };
  });
}

function removeTree(
  nodes: BuilderNode[],
  id: string,
): {
  nodes: BuilderNode[];
  removed: BuilderNode | null;
} {
  const index = nodes.findIndex((node) => node.id === id);

  if (index !== -1) {
    const next = [...nodes];
    const [removed] = next.splice(index, 1);

    return {
      nodes: next,
      removed,
    };
  }

  let removed: BuilderNode | null = null;

  const next = nodes.map((node) => {
    const result = removeTree(node.children || [], id);

    if (result.removed) {
      removed = result.removed;

      return {
        ...node,
        children: result.nodes,
      };
    }

    return node;
  });

  return {
    nodes: next,
    removed,
  };
}

function insertTree(
  nodes: BuilderNode[],
  parentId: string | null,
  node: BuilderNode,
): BuilderNode[] {
  if (!parentId) {
    return [...nodes, node];
  }

  return nodes.map((item) => {
    if (item.id === parentId) {
      return {
        ...item,
        children: [...(item.children || []), node],
      };
    }

    return {
      ...item,
      children: insertTree(item.children || [], parentId, node),
    };
  });
}

function getAllNodeIds(nodes: BuilderNode[]): string[] {
  return nodes.flatMap((node) => [
    node.id,
    ...getAllNodeIds(node.children || []),
  ]);
}

function replaceSiblingList(
  nodes: BuilderNode[],
  parentId: string | null,
  siblings: BuilderNode[],
): BuilderNode[] {
  if (parentId === null) return siblings;

  return updateTree(nodes, parentId, (node) => ({
    ...node,
    children: siblings,
  }));
}

function insertBeforeNode(
  nodes: BuilderNode[],
  parentId: string | null,
  overId: string,
  newNode: BuilderNode,
): BuilderNode[] {
  const siblings =
    parentId === null ? nodes : findNode(nodes, parentId)?.children || [];

  const index = siblings.findIndex((item) => item.id === overId);

  if (index === -1) {
    return insertTree(nodes, parentId, newNode);
  }

  const next = [...siblings];

  next.splice(index, 0, newNode);

  return replaceSiblingList(nodes, parentId, next);
}

function moveNode(
  root: BuilderNode[],
  activeId: string,
  overId: string,
  parentId: string | null,
): BuilderNode[] {
  const activeNode = findNode(root, activeId);

  if (!activeNode || activeId === overId) {
    return root;
  }

  const descendants = getAllNodeIds([activeNode]);

  if (parentId && descendants.includes(parentId)) {
    return root;
  }

  const sourceParent = findParent(root, activeId);

  if (sourceParent === undefined) {
    return root;
  }

  // Same parent: reorder from the original array.
  if (sourceParent === parentId && overId) {
    const siblings =
      parentId === null ? root : findNode(root, parentId)?.children || [];

    const oldIndex = siblings.findIndex((item) => item.id === activeId);

    const newIndex = siblings.findIndex((item) => item.id === overId);

    if (oldIndex === -1 || newIndex === -1) {
      return root;
    }

    return replaceSiblingList(
      root,
      parentId,
      arrayMove(siblings, oldIndex, newIndex),
    );
  }

  const removed = removeTree(root, activeId);

  if (!removed.removed) {
    return root;
  }

  if (overId) {
    const targetParent = findParent(removed.nodes, overId);

    if (targetParent === undefined || targetParent !== parentId) {
      return root;
    }

    return insertBeforeNode(removed.nodes, parentId, overId, removed.removed);
  }

  return insertTree(removed.nodes, parentId, removed.removed);
}

function getResponsiveValue<T>(
  value:
    | {
        desktop?: T;
        tablet?: T;
        mobile?: T;
      }
    | undefined,
  device: BuilderDevice,
): T | undefined {
  if (!value) return undefined;

  return value[device] ?? value.desktop;
}

function getNodeText(node: BuilderNode): string {
  const props = node.props;

  if (typeof props.text === "string") {
    return props.text;
  }

  if (typeof props.content === "string") {
    return props.content;
  }

  if (typeof props.title === "string") {
    return props.title;
  }

  return node.label;
}

function safeMediaUrl(value: string): string {
  const url = value.trim();

  if (!url || /^(javascript|vbscript):/i.test(url)) {
    return "";
  }

  if (
    url.startsWith("/") ||
    url.startsWith("https://") ||
    url.startsWith("http://") ||
    url.startsWith("data:image/")
  ) {
    return url;
  }

  return "";
}

function backgroundToCss(background?: BuilderBackground): CSSProperties {
  if (!background) return {};

  const result: CSSProperties = {};

  if (background.color) {
    result.backgroundColor = background.color;
  }

  if (background.type === "gradient") {
    const gradient = background.gradient;

    if (gradient?.stops?.length) {
      const stops = gradient.stops
        .map((stop) => `${stop.color} ${stop.position}%`)
        .join(", ");

      result.backgroundImage =
        gradient.type === "radial"
          ? `radial-gradient(${gradient.position || "center"}, ${stops})`
          : `linear-gradient(${gradient.angle ?? 180}deg, ${stops})`;
    } else if (gradient) {
      result.backgroundImage = `linear-gradient(${gradient.angle ?? 135}deg, ${gradient.color1 || "#3984e8"}, ${gradient.color2 || "#0c2f55"})`;
    }
  }

  if (background.type === "image" && background.image) {
    const url = safeMediaUrl(background.image);

    if (url) {
      if (background.overlayColor && (background.overlayOpacity ?? 0) > 0) {
        const opacity = Math.max(
          0,
          Math.min(1, background.overlayOpacity || 0),
        );

        result.backgroundImage = `linear-gradient(rgba(0,0,0,${opacity}), rgba(0,0,0,${opacity})), url(${JSON.stringify(url)})`;
      } else {
        result.backgroundImage = `url(${JSON.stringify(url)})`;
      }
    }
  }

  if (background.position) {
    result.backgroundPosition = background.position;
  }

  if (background.positionX || background.positionY) {
    result.backgroundPosition = `${background.positionX?.desktop || "50%"} ${background.positionY?.desktop || "50%"}`;
  }

  if (background.size) {
    result.backgroundSize = background.size;
  }

  if (background.repeat) {
    result.backgroundRepeat =
      background.repeat as CSSProperties["backgroundRepeat"];
  }

  if (background.attachment) {
    result.backgroundAttachment =
      background.attachment as CSSProperties["backgroundAttachment"];
  }

  if (background.blendMode) {
    result.backgroundBlendMode =
      background.blendMode as CSSProperties["backgroundBlendMode"];
  }

  return result;
}

function getNodeInlineStyle(
  node: BuilderNode,
  device: BuilderDevice,
): CSSProperties {
  const style: CSSProperties = {};
  const s = node.styles;

  const put = (key: string, value: unknown) => {
    if (value !== undefined && value !== null && value !== "") {
      (style as unknown as Record<string, unknown>)[key] = value;
    }
  };

  put("width", getResponsiveValue(s.width, device));
  put("height", getResponsiveValue(s.height, device));
  put("minWidth", getResponsiveValue(s.minWidth, device));
  put("maxWidth", getResponsiveValue(s.maxWidth, device));
  put("minHeight", getResponsiveValue(s.minHeight, device));
  put("maxHeight", getResponsiveValue(s.maxHeight, device));

  put("display", getResponsiveValue(s.display, device));
  put("flexDirection", getResponsiveValue(s.direction, device));
  put("justifyContent", getResponsiveValue(s.justifyContent, device));
  put("alignItems", getResponsiveValue(s.alignItems, device));
  put("alignContent", getResponsiveValue(s.alignContent, device));
  put("alignSelf", getResponsiveValue(s.alignSelf, device));
  put("flexWrap", getResponsiveValue(s.flexWrap, device));

  put("gap", getResponsiveValue(s.gap, device));
  put("rowGap", getResponsiveValue(s.rowGap, device));
  put("columnGap", getResponsiveValue(s.columnGap, device));

  put("flexGrow", getResponsiveValue(s.flexGrow, device));
  put("flexShrink", getResponsiveValue(s.flexShrink, device));
  put("flexBasis", getResponsiveValue(s.flexBasis, device));
  put("order", getResponsiveValue(s.order, device));

  put("gridTemplateColumns", getResponsiveValue(s.gridTemplateColumns, device));
  put("gridTemplateRows", getResponsiveValue(s.gridTemplateRows, device));
  put("gridAutoColumns", getResponsiveValue(s.gridAutoColumns, device));
  put("gridAutoRows", getResponsiveValue(s.gridAutoRows, device));
  put("gridAutoFlow", getResponsiveValue(s.gridAutoFlow, device));
  put("gridColumn", getResponsiveValue(s.gridColumn, device));
  put("gridRow", getResponsiveValue(s.gridRow, device));

  const padding = getResponsiveValue(s.padding, device);

  if (padding) {
    style.padding = [
      padding.top || "0",
      padding.right || "0",
      padding.bottom || "0",
      padding.left || "0",
    ].join(" ");
  }

  const margin = getResponsiveValue(s.margin, device);

  if (margin) {
    style.margin = [
      margin.top || "0",
      margin.right || "0",
      margin.bottom || "0",
      margin.left || "0",
    ].join(" ");
  }

  put("color", s.color);

  Object.assign(style, backgroundToCss(s.background));

  if (s.border?.style && s.border.style !== "none") {
    style.borderStyle = s.border.style;
    style.borderWidth = s.border.width || "1px";
    style.borderColor = s.border.color || "#dbe3ed";
  }

  put("borderRadius", getResponsiveValue(s.border?.radius, device));

  put("boxShadow", s.boxShadow);
  put("opacity", s.opacity);
  put("zIndex", s.zIndex);
  put("overflow", s.overflow);

  if (s.position && s.position !== "default") {
    style.position = s.position;
  }

  put("top", getResponsiveValue(s.top, device));
  put("right", getResponsiveValue(s.right, device));
  put("bottom", getResponsiveValue(s.bottom, device));
  put("left", getResponsiveValue(s.left, device));

  const typography = s.typography;

  if (typography) {
    put("fontFamily", typography.fontFamily);
    put("fontSize", getResponsiveValue(typography.fontSize, device));
    put("fontWeight", typography.fontWeight);
    put("lineHeight", typography.lineHeight);
    put("letterSpacing", typography.letterSpacing);
    put("wordSpacing", typography.wordSpacing);
    put("fontStyle", typography.fontStyle);
    put("textTransform", typography.textTransform);
    put("textDecoration", typography.textDecoration);
    put("textAlign", getResponsiveValue(typography.textAlign, device));
    put("textShadow", typography.textShadow);
  }

  if (s.transform) {
    style.transform = s.transform;
  } else if (s.transformSettings) {
    const transform = s.transformSettings;

    style.transform = [
      `rotateX(${transform.rotateX || 0}deg)`,
      `rotateY(${transform.rotateY || 0}deg)`,
      `rotateZ(${transform.rotateZ || 0}deg)`,
      `scale(${transform.scaleX ?? 1}, ${transform.scaleY ?? 1})`,
      `translate(${transform.translateX || "0"}, ${transform.translateY || "0"})`,
      `skew(${transform.skewX || 0}deg, ${transform.skewY || 0}deg)`,
    ].join(" ");

    style.transformOrigin = transform.origin || "center center";
  }

  put("transition", s.transition);
  put("transitionDuration", s.transitionDuration);

  if (s.filters) {
    const filter = s.filters;
    const values = [
      filter.blur ? `blur(${filter.blur}px)` : "",
      filter.brightness !== undefined
        ? `brightness(${filter.brightness}%)`
        : "",
      filter.contrast !== undefined ? `contrast(${filter.contrast}%)` : "",
      filter.saturation !== undefined ? `saturate(${filter.saturation}%)` : "",
      filter.hue ? `hue-rotate(${filter.hue}deg)` : "",
      filter.grayscale ? `grayscale(${filter.grayscale}%)` : "",
      filter.sepia ? `sepia(${filter.sepia}%)` : "",
      filter.invert ? `invert(${filter.invert}%)` : "",
    ].filter(Boolean);

    if (values.length) {
      style.filter = values.join(" ");
    }
  }

  const hidden =
    (device === "desktop" && node.advanced?.hideOnDesktop) ||
    (device === "tablet" && node.advanced?.hideOnTablet) ||
    (device === "mobile" && node.advanced?.hideOnMobile);

  if (hidden || !node.enabled) {
    style.display = "none";
  }

  return style;
}

function SpacingControl({
  label,
  value,
  onChange,
}: {
  label: string;
  value?: BuilderSpacing;
  onChange: (value: BuilderSpacing) => void;
}) {
  const spacing: BuilderSpacing = {
    top: value?.top ?? "0px",
    right: value?.right ?? "0px",
    bottom: value?.bottom ?? "0px",
    left: value?.left ?? "0px",
    unit: value?.unit || "px",
    linked: value?.linked !== false,
  };

  const updateSide = (
    side: "top" | "right" | "bottom" | "left",
    nextValue: string,
  ) => {
    if (spacing.linked) {
      onChange({
        ...spacing,
        top: nextValue,
        right: nextValue,
        bottom: nextValue,
        left: nextValue,
      });
      return;
    }

    onChange({
      ...spacing,
      [side]: nextValue,
    });
  };

  return (
    <div style={{ marginBottom: 15 }}>
      <div
        style={{
          display: "flex",
          alignItems: "center",
          justifyContent: "space-between",
          marginBottom: 7,
        }}
      >
        <span
          style={{
            fontSize: 12,
            fontWeight: 600,
            color: "#334155",
          }}
        >
          {label}
        </span>

        <button
          type="button"
          title={spacing.linked ? "Unlink values" : "Link all values"}
          onClick={() =>
            onChange({
              ...spacing,
              linked: !spacing.linked,
            })
          }
          style={{
            ...BUTTON_STYLE,
            padding: "4px 7px",
            color: spacing.linked ? "#2164c5" : "#64748b",
            borderColor: spacing.linked ? "#3984e8" : "#cbd5e1",
          }}
        >
          {spacing.linked ? <Link2 size={13} /> : <Unlink size={13} />}
        </button>
      </div>

      <div
        style={{
          display: "grid",
          gridTemplateColumns: "repeat(4, minmax(0, 1fr))",
          gap: 5,
        }}
      >
        {(
          [
            ["top", "Top"],
            ["right", "Right"],
            ["bottom", "Bottom"],
            ["left", "Left"],
          ] as const
        ).map(([side, sideLabel]) => (
          <label
            key={side}
            style={{
              display: "flex",
              flexDirection: "column",
              gap: 4,
              minWidth: 0,
            }}
          >
            <span
              style={{
                fontSize: 10,
                textAlign: "center",
                color: "#64748b",
              }}
            >
              {sideLabel}
            </span>

            <input
              value={spacing[side] || ""}
              onChange={(event) => updateSide(side, event.target.value)}
              aria-label={`${label} ${sideLabel}`}
              style={{
                ...CONTROL_STYLE,
                padding: "7px 4px",
                textAlign: "center",
                fontSize: 11,
              }}
            />
          </label>
        ))}
      </div>
    </div>
  );
}

/* -------------------------------------------------------
   Canvas Preview
------------------------------------------------------- */

function PreviewNode({
  node,
  device,
}: {
  node: BuilderNode;
  device: BuilderDevice;
}) {
  const [activeTab, setActiveTab] = useState(0);
  const [hovered, setHovered] = useState(false);

  if (!node.enabled) return null;

  const props = node.props;
  const children = node.children || [];

  const inlineStyle = getNodeInlineStyle(node, device);

  if (hovered && node.styles.backgroundHover) {
    Object.assign(inlineStyle, backgroundToCss(node.styles.backgroundHover));
  }

  const advancedClass = node.advanced?.cssClasses || "";

  const commonProps = {
    id: node.advanced?.cssId || undefined,
    className: advancedClass || undefined,
    style: inlineStyle,
    onMouseEnter: () => setHovered(true),
    onMouseLeave: () => setHovered(false),
  };

  if (LAYOUT_TYPES.has(node.type)) {
    const Tag = node.type === "section" ? "section" : "div";

    return (
      <Tag {...commonProps}>
        {children.map((child) => (
          <PreviewNode key={child.id} node={child} device={device} />
        ))}
      </Tag>
    );
  }

  if (node.type === "heading") {
    const level = String(props.level || "h2");

    switch (level) {
      case "h1":
        return <h1 {...commonProps}>{getNodeText(node)}</h1>;
      case "h3":
        return <h3 {...commonProps}>{getNodeText(node)}</h3>;
      case "h4":
        return <h4 {...commonProps}>{getNodeText(node)}</h4>;
      case "h5":
        return <h5 {...commonProps}>{getNodeText(node)}</h5>;
      case "h6":
        return <h6 {...commonProps}>{getNodeText(node)}</h6>;
      default:
        return <h2 {...commonProps}>{getNodeText(node)}</h2>;
    }
  }

  if (node.type === "text") {
    return (
      <p {...commonProps} className={`${styles.previewText} ${advancedClass}`}>
        {getNodeText(node)}
      </p>
    );
  }

  if (node.type === "button") {
    return (
      <a
        {...commonProps}
        href={String(props.url || "#")}
        className={`${styles.previewButton} ${advancedClass}`}
        target={props.target === "_blank" ? "_blank" : undefined}
        rel={props.target === "_blank" ? "noopener noreferrer" : undefined}
      >
        {props.iconPosition === "before" && props.icon ? (
          <span>{String(props.icon)} </span>
        ) : null}

        {getNodeText(node)}

        {props.iconPosition === "after" && props.icon ? (
          <span> {String(props.icon)}</span>
        ) : null}
      </a>
    );
  }

  if (
    node.type === "image" ||
    node.type === "featured-image" ||
    node.type === "site-logo"
  ) {
    const src =
      typeof props.src === "string"
        ? props.src
        : typeof props.image === "string"
          ? props.image
          : "";

    const imageUrl = safeMediaUrl(src);

    const image = imageUrl ? (
      // eslint-disable-next-line @next/next/no-img-element
      <img
        src={imageUrl}
        alt={String(props.alt || "")}
        title={String(props.title || "")}
        className={styles.previewImage}
        style={{
          ...inlineStyle,
          objectFit: String(
            props.objectFit || "cover",
          ) as CSSProperties["objectFit"],
          maxWidth: "100%",
        }}
      />
    ) : (
      <div className={styles.imagePlaceholder}>
        <ImageIcon size={24} />
        <span>Select an image</span>
      </div>
    );

    return (
      <figure
        {...commonProps}
        style={{
          margin: 0,
          textAlign: String(
            props.alignment || "left",
          ) as CSSProperties["textAlign"],
        }}
      >
        {props.link ? (
          <a
            href={String(props.link)}
            target={props.linkTarget === "_blank" ? "_blank" : undefined}
            rel="noreferrer"
          >
            {image}
          </a>
        ) : (
          image
        )}

        {props.caption ? (
          <figcaption>{String(props.caption)}</figcaption>
        ) : null}
      </figure>
    );
  }

  if (node.type === "divider") {
    return (
      <hr
        {...commonProps}
        className={styles.previewDivider}
        style={{
          ...inlineStyle,
          border: 0,
          borderTop: `${String(props.thickness || "1px")} ${String(props.style || "solid")} ${String(props.color || "#e2e8f0")}`,
        }}
      />
    );
  }

  if (node.type === "spacer") {
    return (
      <div
        {...commonProps}
        className={styles.previewSpacer}
        style={{
          ...inlineStyle,
          minHeight: String(props.height || "40px"),
        }}
      />
    );
  }

  if (node.type === "icon") {
    return (
      <span
        {...commonProps}
        className={styles.previewIcon}
        style={{
          ...inlineStyle,
          fontSize: Number(props.size || 24),
          color: String(props.color || "#0c2f55"),
        }}
      >
        {String(props.symbol || "★")}
      </span>
    );
  }

  if (node.type === "video") {
    const url = safeMediaUrl(String(props.url || ""));

    if (!url) {
      return (
        <div className={styles.imagePlaceholder}>
          <span>Select a video URL</span>
        </div>
      );
    }

    return (
      <video
        {...commonProps}
        src={url}
        poster={
          typeof props.poster === "string"
            ? safeMediaUrl(props.poster)
            : undefined
        }
        controls={props.controls !== false}
        autoPlay={Boolean(props.autoplay)}
        muted={Boolean(props.muted)}
        loop={Boolean(props.loop)}
        playsInline
        style={{
          ...inlineStyle,
          width: "100%",
          maxWidth: "100%",
        }}
      />
    );
  }

  if (node.type === "audio") {
    return (
      <audio
        {...commonProps}
        src={String(props.url || "")}
        controls={props.controls !== false}
        style={{
          width: "100%",
        }}
      />
    );
  }

  if (node.type === "icon-box") {
    return (
      <div
        {...commonProps}
        style={{
          ...inlineStyle,
          textAlign: "center",
        }}
      >
        <div style={{ fontSize: 30 }}>{String(props.symbol || "★")}</div>
        <h3>{String(props.title || "Title")}</h3>
        <p>{String(props.description || "")}</p>
      </div>
    );
  }

  if (node.type === "image-box") {
    return (
      <div {...commonProps}>
        {props.image ? (
          // eslint-disable-next-line @next/next/no-img-element
          <img
            src={safeMediaUrl(String(props.image))}
            alt=""
            style={{
              display: "block",
              maxWidth: "100%",
            }}
          />
        ) : (
          <div className={styles.imagePlaceholder}>
            <ImageIcon size={22} />
          </div>
        )}
        <h3>{String(props.title || "Image Box")}</h3>
        <p>{String(props.description || "")}</p>
      </div>
    );
  }

  if (node.type === "site-title") {
    return <strong {...commonProps}>{getNodeText(node)}</strong>;
  }

  if (node.type === "menu") {
    const menuItems = Array.isArray(props.items)
      ? (props.items as Array<{
          id?: string;
          label?: string;
          title?: string;
          url?: string;
        }>)
      : [];

    return (
      <nav
        {...commonProps}
        className={styles.previewMenu}
        style={{
          ...inlineStyle,
          display: "flex",
          flexWrap: "wrap",
          gap: 18,
        }}
      >
        {menuItems.map((item, index) => (
          <a key={item.id || index} href={item.url || "#"}>
            {item.label || item.title || "Menu item"}
          </a>
        ))}

        {menuItems.length === 0 && (
          <span>Menu preview · {String(props.menuLocation || "HEADER")}</span>
        )}
      </nav>
    );
  }

  if (node.type === "accordion") {
    const items = Array.isArray(props.items)
      ? (props.items as Array<{
          id?: string;
          title?: string;
          content?: string;
        }>)
      : [];

    return (
      <div {...commonProps}>
        {items.map((item, index) => (
          <details
            key={item.id || index}
            open={Boolean(props.firstOpen) && index === 0}
            style={{
              borderBottom: "1px solid #e2e8f0",
              padding: "12px 0",
            }}
          >
            <summary
              style={{
                cursor: "pointer",
                fontWeight: 600,
              }}
            >
              {item.title || `Item ${index + 1}`}
            </summary>
            <div style={{ paddingTop: 9 }}>{item.content || ""}</div>
          </details>
        ))}
      </div>
    );
  }

  if (node.type === "toggle") {
    return (
      <details {...commonProps} open={Boolean(props.open)}>
        <summary
          style={{
            cursor: "pointer",
            fontWeight: 600,
          }}
        >
          {String(props.title || "Toggle")}
        </summary>
        <div style={{ paddingTop: 10 }}>{String(props.content || "")}</div>
      </details>
    );
  }

  if (node.type === "tabs") {
    const tabs = Array.isArray(props.tabs)
      ? (props.tabs as Array<{
          id?: string;
          title?: string;
          content?: string;
        }>)
      : [];

    return (
      <div {...commonProps}>
        <div
          style={{
            display: "flex",
            flexWrap: "wrap",
            gap: 5,
            borderBottom: "1px solid #e2e8f0",
          }}
        >
          {tabs.map((tab, index) => (
            <button
              key={tab.id || index}
              type="button"
              onClick={() => setActiveTab(index)}
              style={{
                ...BUTTON_STYLE,
                borderColor: activeTab === index ? "#3984e8" : "#cbd5e1",
                color: activeTab === index ? "#2164c5" : "#334155",
              }}
            >
              {tab.title || `Tab ${index + 1}`}
            </button>
          ))}
        </div>

        <div style={{ padding: 12 }}>{tabs[activeTab]?.content || ""}</div>
      </div>
    );
  }

  if (node.type === "form") {
    const fields = Array.isArray(props.fields)
      ? (props.fields as Array<{
          id?: string;
          type?: string;
          label?: string;
          name?: string;
          required?: boolean;
          placeholder?: string;
        }>)
      : [];

    return (
      <form {...commonProps} onSubmit={(event) => event.preventDefault()}>
        {fields.map((field, index) => (
          <label
            key={field.id || index}
            style={{
              display: "block",
              marginBottom: 12,
              fontSize: 13,
            }}
          >
            <span>{field.label || "Field"}</span>

            {field.type === "textarea" ? (
              <textarea
                placeholder={field.placeholder}
                required={Boolean(field.required)}
                style={{
                  ...CONTROL_STYLE,
                  display: "block",
                  marginTop: 5,
                  minHeight: 75,
                }}
              />
            ) : (
              <input
                type={
                  ["email", "tel", "number", "url", "password"].includes(
                    String(field.type),
                  )
                    ? String(field.type)
                    : "text"
                }
                name={field.name}
                placeholder={field.placeholder}
                required={Boolean(field.required)}
                style={{
                  ...CONTROL_STYLE,
                  display: "block",
                  marginTop: 5,
                }}
              />
            )}
          </label>
        ))}

        <button
          type="submit"
          style={{
            ...BUTTON_STYLE,
            background: "#0c2f55",
            color: "#fff",
          }}
        >
          {String(props.submitText || "Submit")}
        </button>
      </form>
    );
  }

  if (node.type === "counter") {
    return (
      <div {...commonProps}>
        <strong style={{ fontSize: 28 }}>
          {String(props.prefix || "")}
          {String(props.end || 0)}
          {String(props.suffix || "")}
        </strong>
        <div>{String(props.title || "")}</div>
      </div>
    );
  }

  if (node.type === "progress") {
    const percent = Math.min(100, Math.max(0, Number(props.percentage || 0)));

    return (
      <div {...commonProps}>
        <div>{String(props.title || "Progress")}</div>
        <div
          style={{
            height: 9,
            borderRadius: 20,
            background: "#e2e8f0",
            overflow: "hidden",
          }}
        >
          <div
            style={{
              height: "100%",
              width: `${percent}%`,
              background: "#3984e8",
            }}
          />
        </div>
        <small>{percent}%</small>
      </div>
    );
  }

  if (node.type === "rating") {
    const rating = Math.min(5, Math.max(0, Number(props.rating || 0)));

    return (
      <div {...commonProps}>
        {"★".repeat(Math.round(rating))}
        {"☆".repeat(5 - Math.round(rating))}
      </div>
    );
  }

  if (node.type === "testimonial") {
    return (
      <blockquote {...commonProps}>
        <p>{String(props.quote || "")}</p>
        <strong>{String(props.name || "")}</strong>
        <small>{String(props.role || "")}</small>
      </blockquote>
    );
  }

  if (node.type === "price-table") {
    const features = Array.isArray(props.features) ? props.features : [];

    return (
      <div
        {...commonProps}
        style={{
          ...inlineStyle,
          border: "1px solid #e2e8f0",
          borderRadius: 9,
          padding: 20,
          textAlign: "center",
        }}
      >
        <h3>{String(props.title || "Plan")}</h3>
        <strong style={{ fontSize: 30 }}>
          {String(props.currency || "₹")}
          {String(props.price || "0")}
        </strong>
        <small> / {String(props.period || "month")}</small>

        <ul>
          {features.map((feature, index) => (
            <li key={index}>{String(feature)}</li>
          ))}
        </ul>

        <a href={String(props.buttonUrl || "#")}>
          {String(props.buttonText || "Choose Plan")}
        </a>
      </div>
    );
  }

  if (node.type === "call-to-action") {
    return (
      <div {...commonProps}>
        <h3>{String(props.title || "")}</h3>
        <p>{String(props.description || "")}</p>
        <a href={String(props.buttonUrl || "#")}>
          {String(props.buttonText || "Learn More")}
        </a>
      </div>
    );
  }

  if (node.type === "image-carousel" || node.type === "media-carousel") {
    const rawItems =
      node.type === "image-carousel" ? props.images : props.items;

    const items = Array.isArray(rawItems)
      ? (rawItems as Array<
          string | { url?: string; src?: string; alt?: string }
        >)
      : [];

    return (
      <div
        {...commonProps}
        style={{
          ...inlineStyle,
          display: "flex",
          gap: 12,
          overflowX: "auto",
        }}
      >
        {items.map((item, index) => {
          const url =
            typeof item === "string" ? item : item.url || item.src || "";

          return url ? (
            // eslint-disable-next-line @next/next/no-img-element
            <img
              key={index}
              src={safeMediaUrl(url)}
              alt={typeof item === "string" ? "" : item.alt || ""}
              style={{
                width: "100%",
                maxWidth: 240,
                objectFit: "cover",
              }}
            />
          ) : null;
        })}

        {items.length === 0 && (
          <div className={styles.imagePlaceholder}>Add carousel images</div>
        )}
      </div>
    );
  }

  if (node.type === "posts" || node.type === "loop-grid") {
    return (
      <div
        {...commonProps}
        style={{
          ...inlineStyle,
          display: "grid",
          gridTemplateColumns: `repeat(${Math.max(
            1,
            Number(props.columns || 3),
          )}, minmax(0, 1fr))`,
          gap: 12,
        }}
      >
        {Array.from(
          {
            length: Math.min(6, Math.max(1, Number(props.count || 3))),
          },
          (_, index) => (
            <article
              key={index}
              style={{
                border: "1px solid #e2e8f0",
                borderRadius: 6,
                overflow: "hidden",
                padding: 10,
              }}
            >
              <div
                style={{
                  background: "#e2e8f0",
                  height: 70,
                  marginBottom: 8,
                }}
              />
              <strong>Post preview {index + 1}</strong>
              <p style={{ fontSize: 12 }}>Dynamic post content preview.</p>
            </article>
          ),
        )}
      </div>
    );
  }

  if (node.type === "breadcrumbs") {
    return (
      <nav {...commonProps}>
        Home {String(props.separator || "/")} Current Page
      </nav>
    );
  }

  if (node.type === "google-maps") {
    return (
      <div {...commonProps} className={styles.previewPlaceholder}>
        Map preview: {String(props.address || "Set a map address")}
      </div>
    );
  }

  if (node.type === "lottie") {
    return (
      <div className={styles.previewPlaceholder}>
        Lottie preview: {String(props.url || "Select Lottie JSON")}
      </div>
    );
  }

  if (node.type === "html") {
    return (
      <div className={styles.previewPlaceholder}>
        HTML widget preview. Custom HTML is not executed in the editor canvas.
      </div>
    );
  }

  if (node.type === "shortcode") {
    return (
      <div className={styles.previewPlaceholder}>
        Shortcode: {String(props.shortcode || "Enter shortcode")}
      </div>
    );
  }

  if (node.type === "site-title") {
    return <strong {...commonProps}>{getNodeText(node)}</strong>;
  }

  return (
    <div {...commonProps} className={styles.previewPlaceholder}>
      {getNodeText(node)}
    </div>
  );
}

function PaletteWidget({
  widget,
  onAdd,
}: {
  widget: (typeof BUILDER_WIDGETS)[number];
  onAdd: () => void;
}) {
  const drag = useDraggable({
    id: `palette:${widget.type}`,
    data: {
      kind: "widget",
      widgetType: widget.type,
    },
  });

  return (
    <button
      ref={drag.setNodeRef}
      type="button"
      className={styles.widgetButton}
      style={{
        opacity: drag.isDragging ? 0.4 : 1,
        transform: CSS.Transform.toString(drag.transform),
        touchAction: "none",
      }}
      {...drag.listeners}
      {...drag.attributes}
      onClick={onAdd}
      title={widget.description}
    >
      <span className={styles.widgetIcon}>
        <LayoutGrid size={18} />
      </span>
      <span>{widget.label}</span>
    </button>
  );
}

function DropArea({
  id,
  children,
  className,
}: {
  id: string;
  children: ReactNode;
  className?: string;
}) {
  const { setNodeRef, isOver } = useDroppable({
    id: `drop:${id}`,
    data: {
      kind: "container",
      parentId: id === "root" ? null : id,
    },
  });

  return (
    <div
      ref={setNodeRef}
      className={className}
      data-over={isOver ? "true" : "false"}
      style={{
        minHeight: 30,
        outline: isOver ? "2px dashed #3984e8" : undefined,
        outlineOffset: 3,
        transition: "outline .12s ease",
      }}
    >
      {children}
    </div>
  );
}

function SortableCanvasNode({
  node,
  selectedId,
  device,
  onSelect,
  onDelete,
  onDuplicate,
}: {
  node: BuilderNode;
  selectedId: string | null;
  device: BuilderDevice;
  onSelect: (id: string) => void;
  onDelete: (id: string) => void;
  onDuplicate: (node: BuilderNode) => void;
}) {
  const locked = Boolean(node.locked || node.advanced?.locked);

  const sortable = useSortable({
    id: node.id,
    disabled: locked,
    data: {
      kind: "node",
      nodeId: node.id,
    },
  });

  const selected = selectedId === node.id;
  const children = node.children || [];
  const isLayout = LAYOUT_TYPES.has(node.type);

  return (
    <div
      ref={sortable.setNodeRef}
      className={`${styles.canvasNode} ${
        selected ? styles.canvasNodeSelected : ""
      } ${sortable.isDragging ? styles.canvasNodeDragging : ""}`}
      style={{
        transform: CSS.Transform.toString(sortable.transform),
        transition: sortable.transition,
        opacity: node.enabled ? 1 : 0.45,
      }}
      onClick={(event) => {
        event.stopPropagation();
        onSelect(node.id);
      }}
    >
      <div className={styles.nodeToolbar}>
        <button
          type="button"
          className={styles.dragHandle}
          title={locked ? "Element locked" : "Drag element"}
          disabled={locked}
          {...sortable.attributes}
          {...sortable.listeners}
        >
          <Move size={13} />
        </button>

        <span>
          {node.label}
          {locked ? " 🔒" : ""}
          {!node.enabled ? " (Hidden)" : ""}
        </span>

        <div className={styles.nodeActions}>
          <button
            type="button"
            title="Duplicate"
            onClick={(event) => {
              event.stopPropagation();
              onDuplicate(node);
            }}
          >
            <Copy size={13} />
          </button>

          <button
            type="button"
            title="Delete"
            onClick={(event) => {
              event.stopPropagation();
              onDelete(node.id);
            }}
          >
            <Trash2 size={13} />
          </button>
        </div>
      </div>

      {isLayout ? (
        <div
          className={
            node.type === "section" ? styles.previewSection : styles.nodePreview
          }
          style={getNodeInlineStyle(node, device)}
        >
          <DropArea id={node.id} className={styles.nestedDropArea}>
            {children.length > 0 ? (
              <SortableContext
                items={children.map((child) => child.id)}
                strategy={verticalListSortingStrategy}
              >
                <div className={styles.canvasElements}>
                  {children.map((child) => (
                    <SortableCanvasNode
                      key={child.id}
                      node={child}
                      selectedId={selectedId}
                      device={device}
                      onSelect={onSelect}
                      onDelete={onDelete}
                      onDuplicate={onDuplicate}
                    />
                  ))}
                </div>
              </SortableContext>
            ) : (
              <div className={styles.emptyContainer}>Drop elements here</div>
            )}
          </DropArea>
        </div>
      ) : (
        <div className={styles.nodePreview}>
          <PreviewNode node={node} device={device} />
        </div>
      )}
    </div>
  );
}

function NavigatorNode({
  node,
  depth,
  selectedId,
  onSelect,
  onDelete,
  onDuplicate,
  onToggleEnabled,
  onToggleLocked,
  onRename,
}: {
  node: BuilderNode;
  depth: number;
  selectedId: string | null;
  onSelect: (id: string) => void;
  onDelete: (id: string) => void;
  onDuplicate: (node: BuilderNode) => void;
  onToggleEnabled: (id: string) => void;
  onToggleLocked: (id: string) => void;
  onRename: (node: BuilderNode) => void;
}) {
  const [expanded, setExpanded] = useState(true);

  const locked = Boolean(node.locked || node.advanced?.locked);

  const sortable = useSortable({
    id: `navigator:${node.id}`,
    disabled: locked,
    data: {
      kind: "node",
      nodeId: node.id,
    },
  });

  const hasChildren = Boolean(node.children?.length);

  return (
    <div
      ref={sortable.setNodeRef}
      style={{
        transform: CSS.Transform.toString(sortable.transform),
        transition: sortable.transition,
        opacity: sortable.isDragging ? 0.4 : 1,
      }}
    >
      <div
        className={
          selectedId === node.id
            ? styles.navigatorItemActive
            : styles.navigatorItem
        }
        style={{
          paddingLeft: 8 + depth * 14,
          display: "flex",
          alignItems: "center",
          gap: 3,
          opacity: node.enabled ? 1 : 0.55,
        }}
      >
        <button
          type="button"
          title="Drag to reorder"
          disabled={locked}
          {...sortable.attributes}
          {...sortable.listeners}
          style={{
            border: 0,
            background: "transparent",
            color: "inherit",
            padding: 2,
            cursor: locked ? "not-allowed" : "grab",
          }}
        >
          <Move size={12} />
        </button>

        <button
          type="button"
          onClick={() => setExpanded((value) => !value)}
          aria-label="Expand element"
          style={{
            border: 0,
            background: "transparent",
            padding: 2,
            color: "inherit",
          }}
        >
          {hasChildren ? (
            expanded ? (
              <ChevronDown size={13} />
            ) : (
              <ChevronRight size={13} />
            )
          ) : (
            <span style={{ display: "inline-block", width: 13 }} />
          )}
        </button>

        <button
          type="button"
          onClick={() => onSelect(node.id)}
          onDoubleClick={() => onRename(node)}
          title="Select or double-click to rename"
          style={{
            flex: 1,
            minWidth: 0,
            border: 0,
            background: "transparent",
            color: "inherit",
            textAlign: "left",
            display: "flex",
            alignItems: "center",
            gap: 5,
            fontSize: 12,
          }}
        >
          <Layers size={12} />
          <span
            style={{
              overflow: "hidden",
              textOverflow: "ellipsis",
              whiteSpace: "nowrap",
            }}
          >
            {node.label}
          </span>
        </button>

        <button
          type="button"
          title={node.enabled ? "Hide element" : "Show element"}
          onClick={() => onToggleEnabled(node.id)}
          style={{
            border: 0,
            background: "transparent",
            color: "inherit",
            padding: 2,
            cursor: "pointer",
          }}
        >
          {node.enabled ? <Eye size={12} /> : <EyeOff size={12} />}
        </button>

        <button
          type="button"
          title={locked ? "Unlock" : "Lock"}
          onClick={() => onToggleLocked(node.id)}
          style={{
            border: 0,
            background: "transparent",
            color: "inherit",
            padding: 2,
            cursor: "pointer",
          }}
        >
          {locked ? <Lock size={12} /> : <Unlock size={12} />}
        </button>

        <button
          type="button"
          title="Duplicate"
          onClick={() => onDuplicate(node)}
          style={{
            border: 0,
            background: "transparent",
            color: "inherit",
            padding: 2,
            cursor: "pointer",
          }}
        >
          <Copy size={12} />
        </button>

        <button
          type="button"
          title="Delete"
          onClick={() => onDelete(node.id)}
          style={{
            border: 0,
            background: "transparent",
            color: "inherit",
            padding: 2,
            cursor: "pointer",
          }}
        >
          <Trash2 size={12} />
        </button>
      </div>

      {expanded && node.children && node.children.length > 0 && (
        <SortableContext
          items={node.children.map((child) => `navigator:${child.id}`)}
          strategy={verticalListSortingStrategy}
        >
          {node.children.map((child) => (
            <NavigatorNode
              key={child.id}
              node={child}
              depth={depth + 1}
              selectedId={selectedId}
              onSelect={onSelect}
              onDelete={onDelete}
              onDuplicate={onDuplicate}
              onToggleEnabled={onToggleEnabled}
              onToggleLocked={onToggleLocked}
              onRename={onRename}
            />
          ))}
        </SortableContext>
      )}
    </div>
  );
}

export default function UniversalBuilder({ target }: Props) {
  const documentKey = useMemo(
    () => createDocumentKey(target.kind, target),
    [target.kind, target.id, target.slug, target.postTypeSlug],
  );

  const [layout, setLayout] = useState<BuilderLayout>(() =>
    createDefaultLayout(target.kind),
  );

  const [title, setTitle] = useState(
    target.title || `${KIND_LABELS[target.kind]} Builder`,
  );

  const [status, setStatus] = useState<"DRAFT" | "PUBLISHED" | "ARCHIVED">(
    "DRAFT",
  );

  const [settings, setSettings] = useState<Record<string, unknown>>(() =>
    createDefaultSettings(target.kind),
  );

  const [globalStyles, setGlobalStyles] = useState<BuilderGlobalStyles>(
    DEFAULT_GLOBAL_STYLES,
  );

  const [conditions, setConditions] = useState<Record<string, unknown>>(
    createDefaultConditions(),
  );

  const [selectedId, setSelectedId] = useState<string | null>(null);

  const [device, setDevice] = useState<BuilderDevice>("desktop");

  const [activePanel, setActivePanel] = useState<
    "widgets" | "navigator" | "settings"
  >("widgets");

  const [elementTab, setElementTab] = useState<"widgets" | "globals" | "yoast">(
    "widgets",
  );

  const [inspectorTab, setInspectorTab] = useState<
    "content" | "layout" | "style" | "advanced"
  >("content");

  const [loading, setLoading] = useState(true);
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState("");
  const [message, setMessage] = useState("");

  const [widgetSearch, setWidgetSearch] = useState("");

  const [history, setHistory] = useState<HistoryState[]>([]);
  const [historyIndex, setHistoryIndex] = useState(-1);

  const [mediaPickerOpen, setMediaPickerOpen] = useState(false);

  const [mediaTarget, setMediaTarget] = useState<MediaTarget>("widget");

  const [mediaLoading, setMediaLoading] = useState(false);

  const [mediaItems, setMediaItems] = useState<MediaItem[]>([]);

  const [mediaSearch, setMediaSearch] = useState("");

  const [revisionOpen, setRevisionOpen] = useState(false);

  const [revisionLoading, setRevisionLoading] = useState(false);

  const [revisions, setRevisions] = useState<RevisionItem[]>([]);

  const [importing, setImporting] = useState(false);

  const sensors = useSensors(
    useSensor(PointerSensor, {
      activationConstraint: {
        distance: 6,
      },
    }),
  );

  const selectedNode = useMemo(
    () => findNode(layout.root, selectedId),
    [layout.root, selectedId],
  );

  const isLayoutNode = Boolean(
    selectedNode && LAYOUT_TYPES.has(selectedNode.type),
  );

  const commitRoot = useCallback(
    (nextRoot: BuilderNode[]) => {
      setLayout({
        version: 1,
        root: nextRoot,
      });

      setHistory((current) => {
        const base = current.slice(0, historyIndex + 1);

        return [
          ...base,
          {
            root: structuredClone(nextRoot),
          },
        ];
      });

      setHistoryIndex((index) => index + 1);
    },
    [historyIndex],
  );

  const undo = () => {
    if (historyIndex <= 0) return;

    const nextIndex = historyIndex - 1;

    setHistoryIndex(nextIndex);

    setLayout({
      version: 1,
      root: structuredClone(history[nextIndex].root),
    });
  };

  const redo = () => {
    if (historyIndex >= history.length - 1) {
      return;
    }

    const nextIndex = historyIndex + 1;

    setHistoryIndex(nextIndex);

    setLayout({
      version: 1,
      root: structuredClone(history[nextIndex].root),
    });
  };

  const loadDocument = useCallback(async () => {
    setLoading(true);
    setError("");

    try {
      const response = await fetch(
        `/api/admin/builder/documents?documentKey=${encodeURIComponent(
          documentKey,
        )}`,
        {
          cache: "no-store",
        },
      );

      const result = (await response.json()) as DocumentResponse;

      if (!response.ok || !result.success) {
        throw new Error(result.message || "Unable to load builder.");
      }

      if (result.document) {
        const doc = result.document;

        const root =
          doc.layout?.version === 1 && Array.isArray(doc.layout.root)
            ? doc.layout.root
            : createDefaultLayout(target.kind).root;

        setLayout({
          version: 1,
          root,
        });

        setHistory([
          {
            root: structuredClone(root),
          },
        ]);

        setHistoryIndex(0);
        setTitle(doc.title);
        setStatus(doc.status);

        setSettings(doc.settings || createDefaultSettings(target.kind));

        setGlobalStyles(doc.globalStyles || DEFAULT_GLOBAL_STYLES);

        setConditions(
          (doc.conditions || createDefaultConditions()) as Record<
            string,
            unknown
          >,
        );
      } else {
        const root = createDefaultLayout(target.kind).root;

        setLayout({
          version: 1,
          root,
        });

        setHistory([
          {
            root: structuredClone(root),
          },
        ]);

        setHistoryIndex(0);
        setStatus("DRAFT");
        setSettings(createDefaultSettings(target.kind));
        setGlobalStyles(DEFAULT_GLOBAL_STYLES);
        setConditions(createDefaultConditions());
      }

      setSelectedId(null);
    } catch (err) {
      setError(
        err instanceof Error ? err.message : "Unable to load builder document.",
      );
    } finally {
      setLoading(false);
    }
  }, [documentKey, target.kind]);

  useEffect(() => {
    void loadDocument();
  }, [loadDocument]);

  const createWidget = (type: string): BuilderNode | null => {
    const definition = BUILDER_WIDGETS.find((item) => item.type === type);

    if (!definition) return null;

    return createDefaultNode(
      definition.type,
      definition.label,
      makeId(definition.type),
      structuredClone(definition.defaultProps),
      [],
    );
  };

  const addWidget = (type: string, parentId: string | null = null) => {
    const node = createWidget(type);

    if (!node) return;

    commitRoot(insertTree(layout.root, parentId, node));

    setSelectedId(node.id);
    setActivePanel("settings");
    setInspectorTab(LAYOUT_TYPES.has(node.type) ? "layout" : "content");
  };

  const deleteNode = (id: string) => {
    const result = removeTree(layout.root, id);

    if (!result.removed) return;

    const deletedIds = getAllNodeIds([result.removed]);

    commitRoot(result.nodes);

    if (selectedId && deletedIds.includes(selectedId)) {
      setSelectedId(null);
    }
  };

  const duplicateNode = (node: BuilderNode) => {
    const duplicate = cloneNode(node);

    duplicate.label = `${node.label} Copy`;

    const parentId = findParent(layout.root, node.id);

    if (parentId === undefined) return;

    const siblings =
      parentId === null
        ? layout.root
        : findNode(layout.root, parentId)?.children || [];

    const index = siblings.findIndex((item) => item.id === node.id);

    if (index === -1) return;

    const next = [...siblings];

    next.splice(index + 1, 0, duplicate);

    const nextRoot =
      parentId === null
        ? next
        : updateTree(layout.root, parentId, (parent) => ({
            ...parent,
            children: next,
          }));

    commitRoot(nextRoot);
    setSelectedId(duplicate.id);
    setActivePanel("settings");
  };

  const renameNode = (node: BuilderNode) => {
    const nextLabel = window.prompt("Element name", node.label);

    if (nextLabel === null || !nextLabel.trim()) {
      return;
    }

    commitRoot(
      updateTree(layout.root, node.id, (current) => ({
        ...current,
        label: nextLabel.trim(),
      })),
    );
  };

  const toggleNodeEnabled = (id: string) => {
    commitRoot(
      updateTree(layout.root, id, (node) => ({
        ...node,
        enabled: !node.enabled,
      })),
    );
  };

  const toggleNodeLocked = (id: string) => {
    commitRoot(
      updateTree(layout.root, id, (node) => {
        const locked = !(node.locked || node.advanced?.locked);

        return {
          ...node,
          locked,
          advanced: {
            ...(node.advanced || {}),
            locked,
          },
        };
      }),
    );
  };

  const handleDragEnd = (event: DragEndEvent) => {
    const { active, over } = event;

    if (!over) return;

    const activeData = active.data.current;
    const overData = over.data.current;

    if (activeData?.kind === "widget") {
      const type = String(activeData.widgetType || "");

      const node = createWidget(type);

      if (!node) return;

      let parentId: string | null = null;
      let overId = "";

      if (overData?.kind === "container") {
        parentId = (overData.parentId as string | null) ?? null;
      } else if (overData?.kind === "node") {
        const targetId = String(overData.nodeId || over.id);

        const targetNode = findNode(layout.root, targetId);

        if (!targetNode) return;

        if (LAYOUT_TYPES.has(targetNode.type)) {
          parentId = targetNode.id;
        } else {
          parentId = findParent(layout.root, targetId) ?? null;
          overId = targetId;
        }
      } else {
        return;
      }

      const nextRoot = overId
        ? insertBeforeNode(layout.root, parentId, overId, node)
        : insertTree(layout.root, parentId, node);

      commitRoot(nextRoot);
      setSelectedId(node.id);
      setActivePanel("settings");
      setInspectorTab(LAYOUT_TYPES.has(node.type) ? "layout" : "content");

      return;
    }

    if (activeData?.kind !== "node") return;

    const activeId = String(
      activeData.nodeId || String(active.id).replace("navigator:", ""),
    );

    const movingNode = findNode(layout.root, activeId);

    if (!movingNode || movingNode.locked || movingNode.advanced?.locked) {
      return;
    }

    let parentId: string | null = null;
    let overId = "";

    if (overData?.kind === "container") {
      parentId = (overData.parentId as string | null) ?? null;
    } else if (overData?.kind === "node") {
      const targetId = String(overData.nodeId || over.id);

      if (targetId === activeId) return;

      const targetNode = findNode(layout.root, targetId);

      if (!targetNode) return;

      if (LAYOUT_TYPES.has(targetNode.type)) {
        parentId = targetNode.id;
      } else {
        parentId = findParent(layout.root, targetId) ?? null;
        overId = targetId;
      }
    } else {
      return;
    }

    const descendants = getAllNodeIds([movingNode]);

    if (parentId && descendants.includes(parentId)) {
      return;
    }

    const nextRoot = moveNode(layout.root, activeId, overId, parentId);

    if (nextRoot !== layout.root) {
      commitRoot(nextRoot);
    }
  };

  const updateProps = (key: string, value: unknown) => {
    if (!selectedId) return;

    commitRoot(
      updateTree(layout.root, selectedId, (node) => ({
        ...node,
        props: {
          ...node.props,
          [key]: value,
        },
      })),
    );
  };

  const updateLabel = (value: string) => {
    if (!selectedId) return;

    commitRoot(
      updateTree(layout.root, selectedId, (node) => ({
        ...node,
        label: value,
      })),
    );
  };

  const updateResponsiveStyle = (key: keyof BuilderStyle, value: unknown) => {
    if (!selectedId) return;

    commitRoot(
      updateTree(layout.root, selectedId, (node) => {
        const current = node.styles[key] as Record<string, unknown> | undefined;

        return {
          ...node,
          styles: {
            ...node.styles,
            [key]: {
              ...(current || {}),
              [device]: value,
            },
          } as BuilderStyle,
        };
      }),
    );
  };

  const updateSimpleStyle = (key: keyof BuilderStyle, value: unknown) => {
    if (!selectedId) return;

    commitRoot(
      updateTree(layout.root, selectedId, (node) => ({
        ...node,
        styles: {
          ...node.styles,
          [key]: value,
        },
      })),
    );
  };

  const updateSpacing = (key: "padding" | "margin", value: BuilderSpacing) => {
    if (!selectedId) return;

    commitRoot(
      updateTree(layout.root, selectedId, (node) => ({
        ...node,
        styles: {
          ...node.styles,
          [key]: {
            ...(node.styles[key] || {}),
            [device]: value,
          },
        },
      })),
    );
  };

  const updateTypography = (
    key: keyof BuilderTypography,
    value: unknown,
    responsive = false,
  ) => {
    if (!selectedId) return;

    commitRoot(
      updateTree(layout.root, selectedId, (node) => {
        const typography = node.styles.typography || {};

        const current = typography[key] as Record<string, unknown> | undefined;

        return {
          ...node,
          styles: {
            ...node.styles,
            typography: {
              ...typography,
              [key]: responsive
                ? {
                    ...(current || {}),
                    [device]: value,
                  }
                : value,
            } as BuilderTypography,
          },
        };
      }),
    );
  };

  const updateBackground = (
    key: keyof BuilderBackground,
    value: unknown,
    hover = false,
  ) => {
    if (!selectedId) return;

    commitRoot(
      updateTree(layout.root, selectedId, (node) => {
        const current = hover
          ? node.styles.backgroundHover || {}
          : node.styles.background || {};

        const nextBackground = {
          ...current,
          [key]: value,
        };

        return {
          ...node,
          styles: {
            ...node.styles,
            ...(hover
              ? {
                  backgroundHover: nextBackground,
                }
              : {
                  background: nextBackground,
                }),
          },
        };
      }),
    );
  };

  const updateBorder = (key: keyof BuilderBorder, value: unknown) => {
    if (!selectedId) return;

    commitRoot(
      updateTree(layout.root, selectedId, (node) => ({
        ...node,
        styles: {
          ...node.styles,
          border: {
            ...(node.styles.border || {}),
            [key]: value,
          },
        },
      })),
    );
  };

  const updateAdvanced = (key: string, value: unknown) => {
    if (!selectedId) return;

    commitRoot(
      updateTree(layout.root, selectedId, (node) => ({
        ...node,
        advanced: {
          ...(node.advanced || {}),
          [key]: value,
        },
      })),
    );
  };

  const openMediaLibrary = async (mediaType: MediaTarget = "widget") => {
    setMediaTarget(mediaType);
    setMediaPickerOpen(true);
    setMediaLoading(true);
    setMediaSearch("");
    setError("");

    try {
      const response = await fetch("/api/admin/media?view=library", {
        cache: "no-store",
      });

      const result = await response.json();

      if (!response.ok) {
        throw new Error(result.message || "Media Library load nahi hui.");
      }

      const items = Array.isArray(result.media)
        ? result.media
        : Array.isArray(result.items)
          ? result.items
          : [];

      setMediaItems(items);
    } catch (err) {
      setError(
        err instanceof Error ? err.message : "Media Library load nahi hui.",
      );
    } finally {
      setMediaLoading(false);
    }
  };

  const chooseMedia = (item: MediaItem) => {
    if (!selectedId) return;

    const mediaUrl = item.filePath || item.url || "";

    if (!mediaUrl) {
      setError("Selected media URL available nahi hai.");
      return;
    }

    commitRoot(
      updateTree(layout.root, selectedId, (node) => {
        if (mediaTarget === "background") {
          return {
            ...node,
            styles: {
              ...node.styles,
              background: {
                ...(node.styles.background || {}),
                type: "image",
                image: mediaUrl,
                imageId: item.id,
              },
            },
          };
        }

        if (mediaTarget === "hoverBackground") {
          return {
            ...node,
            styles: {
              ...node.styles,
              backgroundHover: {
                ...(node.styles.backgroundHover || {}),
                type: "image",
                image: mediaUrl,
                imageId: item.id,
              },
            },
          };
        }

        return {
          ...node,
          props: {
            ...node.props,
            src: mediaUrl,
            image: mediaUrl,
            alt: item.altText || node.props.alt || "",
            title: item.title || node.props.title || "",
            mediaId: item.id,
          },
        };
      }),
    );

    setMediaPickerOpen(false);
  };

  const saveDocument = async (nextStatus: "DRAFT" | "PUBLISHED") => {
    setSaving(true);
    setError("");
    setMessage("");

    try {
      const response = await fetch("/api/admin/builder/documents", {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          documentKey,
          kind: target.kind,
          title,
          slug: target.slug ?? null,
          targetId: target.id ?? null,
          postTypeSlug: target.postTypeSlug ?? null,
          status: nextStatus,
          layout,
          settings,
          globalStyles,
          conditions,
          revisionMessage:
            nextStatus === "PUBLISHED"
              ? `Published from Builder - ${new Date().toLocaleString()}`
              : undefined,
        }),
      });

      const result = await response.json();

      if (!response.ok || !result.success) {
        throw new Error(result.message || "Unable to save builder.");
      }

      setStatus(nextStatus);
      setMessage(result.message || "Saved successfully.");
    } catch (err) {
      setError(err instanceof Error ? err.message : "Unable to save builder.");
    } finally {
      setSaving(false);
    }
  };

  const loadRevisions = async () => {
    setRevisionOpen(true);
    setRevisionLoading(true);
    setError("");

    try {
      const response = await fetch(
        `/api/admin/builder/documents?documentKey=${encodeURIComponent(
          documentKey,
        )}&action=revisions`,
        {
          cache: "no-store",
        },
      );

      const result = await response.json();

      if (!response.ok || !result.success) {
        throw new Error(result.message || "Unable to load revision history.");
      }

      setRevisions(Array.isArray(result.revisions) ? result.revisions : []);
    } catch (err) {
      setError(
        err instanceof Error ? err.message : "Unable to load revisions.",
      );
    } finally {
      setRevisionLoading(false);
    }
  };

  const restoreRevision = async (revisionId: string) => {
    const confirmed = window.confirm(
      "Restore this revision? Your current document will be replaced.",
    );

    if (!confirmed) return;

    setSaving(true);
    setError("");
    setMessage("");

    try {
      const response = await fetch("/api/admin/builder/documents", {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          action: "restore",
          documentKey,
          revisionId,
        }),
      });

      const result = await response.json();

      if (!response.ok || !result.success) {
        throw new Error(result.message || "Unable to restore revision.");
      }

      setRevisionOpen(false);
      await loadDocument();

      setMessage(result.message || "Revision restored successfully.");
    } catch (err) {
      setError(
        err instanceof Error ? err.message : "Unable to restore revision.",
      );
    } finally {
      setSaving(false);
    }
  };

  const exportDocument = () => {
    const data = {
      format: "connect-broadband-builder",
      version: 1,
      exportedAt: new Date().toISOString(),
      kind: target.kind,
      title,
      layout,
      settings,
      globalStyles,
      conditions,
    };

    const blob = new Blob([JSON.stringify(data, null, 2)], {
      type: "application/json",
    });

    const url = URL.createObjectURL(blob);
    const anchor = document.createElement("a");

    anchor.href = url;
    anchor.download = `${
      title
        .toLowerCase()
        .replace(/[^a-z0-9]+/g, "-")
        .replace(/^-|-$/g, "") || "builder-template"
    }.json`;

    anchor.click();

                        borderBottom:
                          elementTab === tab
                            ? "2px solid #3984e8"
                            : "2px solid transparent",
                        background: "transparent",
                        fontWeight: elementTab === tab ? 700 : 400,
                        cursor: "pointer",
                      }}
                    >
                      {tab === "yoast"
                        ? "SEO"
                        : tab === "globals"
                          ? "Globals"
                          : "Widgets"}
                    </button>
                  ))}
                </div>

                {elementTab === "widgets" && (
                  <>
                    <div className={styles.widgetSearch}>
                      <Search size={15} />
                      <input
                        value={widgetSearch}
                        onChange={(event) =>
                          setWidgetSearch(event.target.value)
                        }
                        placeholder="Search elements..."
                      />
                    </div>

                    {widgetGroups.map((group) => (
                      <section
                        className={styles.widgetGroup}
                        key={group.category}
                      >
                        <h3>{group.category}</h3>

                        <div className={styles.widgetGrid}>
                          {group.widgets.map((widget) => (
                            <PaletteWidget
                              key={widget.type}
                              widget={widget}
                              onAdd={() => addWidget(widget.type)}
                            />
                          ))}
                        </div>
                      </section>
                    ))}

                    {widgetGroups.length === 0 && (
                      <p className={styles.mutedText}>No elements found.</p>
                    )}
                  </>
                )}

                {elementTab === "globals" && (
                  <div className={styles.settingsPanel}>
                    <ControlGroup title="Global Colors">
                      {globalStyles.colors.map((color) => (
                        <Field key={color.id} label={color.name}>
                          <div
                            style={{
                              display: "grid",
                              gridTemplateColumns: "48px 1fr",
                              gap: 8,

                              type="color"

                                setGlobalColor(color.id, event.target.value)

                            <input
                              value={color.value}
                              onChange={(event) =>
                                setGlobalColor(color.id, event.target.value)
                              }
                              style={CONTROL_STYLE}
                            />
                          </div>
                        </Field>
                      ))}
                    </ControlGroup>

                    <ControlGroup title="Global Typography">
                      {globalStyles.fonts.map((font) => (

                          <input
                            value={font.family}
                            onChange={(event) =>
                              setGlobalFont(font.id, event.target.value)
                            }
                            style={CONTROL_STYLE}
                          />
                        </Field>
                      ))}
                    </ControlGroup>

                    <ControlGroup title="Theme Defaults">
                      <TextControl
                        label="Container Width"
                        value={String(settings.containerWidth || "1200px")}

                          updateSetting("containerWidth", value)

                      />

                      <TextControl
                        label="Default Widget Gap"
                        value={String(settings.defaultWidgetGap || "10px")}
                        onChange={(value) =>
                          updateSetting("defaultWidgetGap", value)
                        }
                      />

                      <TextControl
                        label="Default Section Padding"
                        value={String(settings.defaultSectionPadding || "20px")}
                        onChange={(value) =>
                          updateSetting("defaultSectionPadding", value)
                        }
                      />
                    </ControlGroup>
                  </div>
                )}

                {elementTab === "yoast" && (
                  <div className={styles.settingsPanel}>
                    <ControlGroup title="SEO">
                      <TextControl
                        label="SEO Title"
                        value={String(settings.seoTitle || "")}
                        onChange={(value) => updateSetting("seoTitle", value)}
                      />

                      <Field label="Meta Description">
                        <textarea
                          rows={5}

                        label="Focus Keyphrase"

                          updateSetting("focusKeyphrase", value)

                    </ControlGroup>
                  </div>
                )}
              </div>
            )}

            {activePanel === "navigator" && (
              <div className={styles.navigatorPanel}>
                <div className={styles.panelHeading}>Navigator / Structure</div>

                {layout.root.length === 0 ? (
                  <p className={styles.mutedText}>No elements added yet.</p>

                          setInspectorTab(
                            selected && LAYOUT_TYPES.has(selected.type)

                        onDuplicate={duplicateNode}

                        onToggleLocked={toggleNodeLocked}

            {activePanel === "settings" && (

                      : ""

                    <div className={styles.emptyCanvas}>
                      <div className={styles.emptyCanvasIcon}>
                        <Plus size={25} />
                      </div>

                      <h2>Start building your {target.kind}</h2>

                      <p>Drag a Section, Container or widget here.</p>

                      <button
                        type="button"
                        onClick={() => {
                          setActivePanel("widgets");
                          setElementTab("widgets");
                        }}
                      >
                        Add Your First Element
                      </button>
                    </div>
                  ) : (
                    <SortableContext
                      items={layout.root.map((node) => node.id)}
                      strategy={verticalListSortingStrategy}
                    >
                      {layout.root.map((node) => (
                        <SortableCanvasNode
                          key={node.id}
                          node={node}
                          selectedId={selectedId}
                          device={device}
                          onSelect={(id) => {
                            const current = findNode(layout.root, id);

                            setSelectedId(id);
                            setActivePanel("settings");
                            setInspectorTab(
                              current && LAYOUT_TYPES.has(current.type)
                                ? "layout"
                                : "content",
                            );
                          }}
                          onDelete={deleteNode}
                          onDuplicate={duplicateNode}
                        />
                      ))}
                    </SortableContext>
                  )}
                </DropArea>
              </div>
            </div>
          </section>
        </div>

        {mediaPickerOpen && (
          <div
            role="dialog"
            aria-modal="true"
            aria-label="Media Library"
            style={{
              position: "fixed",
              inset: 0,
              zIndex: 99999,
              background: "rgba(15,23,42,.65)",
              display: "flex",
              alignItems: "center",
              justifyContent: "center",
              padding: 20,
            }}
          >
            <div
              style={{
                width: "min(1000px, 96vw)",
                height: "min(720px, 90vh)",
                background: "#fff",
                borderRadius: 10,
                display: "flex",
                flexDirection: "column",
                overflow: "hidden",
                boxShadow: "0 20px 70px rgba(0,0,0,.25)",

              <div
                style={{
                  padding: 18,
                  display: "flex",
                  alignItems: "center",
                  justifyContent: "space-between",

                }}

                <div>
                  <strong>Media Library</strong>
                  <div
                    style={{
                      fontSize: 12,
                      color: "#64748b",

                  >
                    Choose media for this element
                  </div>
                </div>

                <button
                  type="button"
                  onClick={() => setMediaPickerOpen(false)}
                  aria-label="Close Media Library"

                  <X size={18} />

                style={{
                  padding: 14,
                  borderBottom: "1px solid #e2e8f0",

                <div className={styles.widgetSearch}>
                  <Search size={15} />

                  <input
                    value={mediaSearch}
                    onChange={(event) => setMediaSearch(event.target.value)}

              </div>

              <div
                style={{
                  flex: 1,
                  overflow: "auto",
                  padding: 16,

              >
                {mediaLoading ? (
                  <div
                    style={{
                      display: "flex",
                      justifyContent: "center",
                      padding: 50,
                    }}
                  >
                    <Loader2 className={styles.spin} size={28} />
                  </div>
                ) : filteredMedia.length === 0 ? (
                  <div
                    style={{

                            {item.originalName || item.fileName || item.id}

                )}

              <div
                style={{
                  padding: 14,
                  borderTop: "1px solid #e2e8f0",
                  display: "flex",
                  justifyContent: "flex-end",

                <button
                  type="button"
                  onClick={() => setMediaPickerOpen(false)}
                  className={styles.previewButton}
                >
                  Cancel
                </button>
              </div>
            </div>

        {revisionOpen && (

            role="dialog"
            aria-modal="true"
            aria-label="Revision History"

              position: "fixed",
              inset: 0,
              zIndex: 99999,

                  style={BUTTON_STYLE}

              <div
                style={{

                {revisionLoading ? (
                  <div
                    style={{
                      textAlign: "center",
                      padding: 30,
                    }}
                  >
                    <Loader2 className={styles.spin} size={24} />
                  </div>
                ) : revisions.length === 0 ? (
                  <p className={styles.mutedText}>
                    No published revisions found yet.
                  </p>
                ) : (
                  revisions.map((revision) => (
                    <div
                      key={revision.id}
                      style={{
                        display: "flex",
                        alignItems: "center",
                        justifyContent: "space-between",
                        gap: 12,
                        borderBottom: "1px solid #e2e8f0",
                        padding: "12px 0",
                      }}
                    >
                      <div>
                        <strong>Revision #{revision.revisionNo}</strong>

                        <div
                          style={{
                            fontSize: 13,
                            color: "#64748b",
                          }}
                        >
                          {revision.message || revision.title}
                        </div>

                        <div
                          style={{
                            fontSize: 12,
                            color: "#94a3b8",
                          }}
                        >
                          {revision.createdAt
                            ? new Date(revision.createdAt).toLocaleString()
                            : ""}
                        </div>
                      </div>

                      <button
                        type="button"
                        className={styles.previewButton}
                        disabled={saving}
                        onClick={() => void restoreRevision(revision.id)}
                      >
                        <RotateCcw size={14} />
                        Restore
                      </button>
                    </div>
                  ))
                )}
              </div>
            </div>
          </div>
        )}
      </main>
    </DndContext>
  );
}

export function createDocumentKey(
  kind: BuilderKind,
  target?: {
    id?: string;
    slug?: string;
    postTypeSlug?: string;
  },
): string {
  switch (kind) {
    case "header":
      return "header:primary";

    case "footer":
      return "footer:primary";

    case "page":
      if (!target?.id) {
        throw new Error("Page ID is required.");
      }
      return `page:${target.id}`;

    case "post":
      if (!target?.id) {
        throw new Error("Post ID is required.");
      }
      return `post:${target.id}`;

    case "single":
      if (!target?.slug) {
        throw new Error("Single template slug is required.");
      }
      return `single:${target.slug}`;

    case "archive":
      if (!target?.slug) {
        throw new Error("Archive template slug is required.");
      }
      return `archive:${target.slug}`;

    case "custom":
      if (!target?.postTypeSlug || !target?.id) {
        throw new Error(
          "Custom post type slug and ID are required.",
        );
      }
      return `custom:${target.postTypeSlug}:${target.id}`;

    default:
      throw new Error("Unsupported builder kind.");
  }
}

function getContentData(content: unknown) {
  if (
    content &&
    typeof content === "object" &&
    !Array.isArray(content)
  ) {
    return content as {
      text?: unknown;
      featuredImage?: unknown;
      layout?: unknown;
    };
  }

  return {};
}

function getPageHtml(content: unknown): string {
  const contentData = getContentData(content);
  const storedText = contentData.text;

  if (typeof storedText !== "string" || !storedText.trim()) {
    return "";
  }

  try {
    const parsed = JSON.parse(storedText);

    if (
      parsed &&
      typeof parsed === "object" &&
      typeof parsed.html === "string"
    ) {
      return parsed.html;
    }
  } catch {
    // Stored content may already be HTML.
  }

  return storedText;
}

function getFeaturedImage(content: unknown): string {
  const contentData = getContentData(content);

  return typeof contentData.featuredImage === "string"
    ? contentData.featuredImage
    : "";
}

async function findPublishedPage(
  slugs: string[]
): Promise<CMSPage | null> {
  if (!slugs.length) {
    return null;
  }

  let parentId: string | null = null;
  let currentPage: CMSPage | null = null;

  for (const slug of slugs) {
    const page: CMSPage | null = await prisma.page.findFirst({
      where: {
  slug,
  parentId,
  status: "PUBLISHED",
  deletedAt: null,
},
      select: {
        id: true,
        title: true,
        slug: true,
        description: true,
        seoTitle: true,
        seoDescription: true,
        content: true,
      },
    });

    if (!page) {
      return null;
    }

    currentPage = page;
    parentId = page.id;
  }

  return currentPage;
}

export async function generateMetadata({
  params,
}: {
  params: Promise<PageParams>;
}): Promise<Metadata> {
  const { slug } = await params;
  const page = await findPublishedPage(slug);

  if (!page) {
    return {
      title: "Page Not Found | Connect Broadband",
    };
  }

  const title =
    page.seoTitle?.trim() ||
    `${page.title} | Connect Broadband`;

  const description =
    page.seoDescription?.trim() ||
    page.description?.trim() ||
    undefined;

  return {
    title,
    description,
    openGraph: {
      title,
      description,
      type: "website",
    },
  };
}

import { NextRequest, NextResponse } from "next/server";
import { Prisma } from "@prisma/client";

import { prisma } from "@/lib/prisma";
import { getCurrentUser } from "@/lib/auth/session";
import { verifyRequestOrigin } from "@/lib/auth/request";

const ALLOWED_KINDS = [
  "header",
  "footer",
  "page",
  "post",
  "single",
  "archive",
  "custom",
] as const;

type BuilderKind = (typeof ALLOWED_KINDS)[number];

type SaveBody = {
  action?: "save" | "restore";
  revisionId?: string;
  documentKey?: string;
  kind?: BuilderKind;
  title?: string;
  slug?: string | null;
  targetId?: string | null;
  postTypeSlug?: string | null;
  status?: "DRAFT" | "PUBLISHED";
  layout?: {
    version: 1;
    root: unknown[];
  };
  settings?: Record<string, unknown>;
  globalStyles?: Record<string, unknown>;
  conditions?: Record<string, unknown>;
  revisionMessage?: string;
};

type Snapshot = {
  layout?: {
    version: 1;
    root: unknown[];
  };
  settings?: Record<string, unknown>;
  globalStyles?: Record<string, unknown>;
  conditions?: Record<string, unknown>;
  status?: "DRAFT" | "PUBLISHED" | "ARCHIVED";
};

function jsonError(
  message: string,
  status = 400,
) {
  return NextResponse.json(
    { success: false, message },
    { status },
  );
}

async function authorize() {
  const user = await getCurrentUser();

  if (!user || !user.isActive) {
    return {
      user: null,
      response: jsonError(
        "Unauthorized. Please log in again.",
        401,
      ),
    };
  }

  return {
    user,
    response: null,
  };
}

function isValidDocumentKey(
  value: unknown,
): value is string {
  return (
    typeof value === "string" &&
    value.length > 0 &&
    value.length <= 191 &&
    /^[a-zA-Z0-9:_-]+$/.test(value)
  );
}

function countNodes(
  nodes: unknown[],
  depth = 0,
  seen = new Set<string>(),
): number {
  if (depth > 20) {
    throw new Error(
      "Builder layout nesting limit exceeded.",
    );
  }

  let count = 0;

  for (const item of nodes) {
    if (
      !item ||
      typeof item !== "object" ||
      Array.isArray(item)
    ) {
      throw new Error("Invalid builder element.");
    }

    const node = item as Record<string, unknown>;

    if (
      typeof node.id !== "string" ||
      !node.id.trim() ||
      typeof node.type !== "string" ||
      !node.type.trim() ||
      typeof node.label !== "string" ||
      typeof node.enabled !== "boolean" ||
      !node.props ||
      typeof node.props !== "object" ||
      Array.isArray(node.props) ||
      !node.styles ||
      typeof node.styles !== "object" ||
      Array.isArray(node.styles)
    ) {
      throw new Error(
        "Builder element has an invalid structure.",
      );
    }

    if (seen.has(node.id)) {
      throw new Error(
        `Duplicate builder element ID: ${node.id}`,
      );
    }

    seen.add(node.id);
    count += 1;

    if (Array.isArray(node.children)) {
      count += countNodes(
        node.children,
        depth + 1,
        seen,
      );
    }

    if (count > 500) {
      throw new Error(
        "Builder layout cannot contain more than 500 elements.",
      );
    }
  }

  return count;
}

function isValidLayout(
  value: unknown,
): value is {
  version: 1;
  root: unknown[];
} {
  if (
    !value ||
    typeof value !== "object" ||
    Array.isArray(value)
  ) {
    return false;
  }

  const layout = value as Record<string, unknown>;

  if (
    layout.version !== 1 ||
    !Array.isArray(layout.root)
  ) {
    return false;
  }

  countNodes(layout.root);

  return true;
}

export async function GET(request: NextRequest) {
  const auth = await authorize();

  if (auth.response) return auth.response;

  const documentKey =
    request.nextUrl.searchParams.get("documentKey");

  const action =
    request.nextUrl.searchParams.get("action");

  if (!isValidDocumentKey(documentKey)) {
    return jsonError(
      "A valid documentKey is required.",
    );
  }

  try {
    const document =
      await prisma.builderDocument.findUnique({
        where: { documentKey },
      });

    if (action === "revisions") {
      if (!document) {
        return NextResponse.json({
          success: true,
          revisions: [],
        });
      }

      const revisions =
        await prisma.builderRevision.findMany({
          where: {
            documentId: document.id,
          },
          orderBy: {
            revisionNo: "desc",
          },
          take: 100,
        });

      return NextResponse.json({
        success: true,
        revisions,
      });
    }

    return NextResponse.json({
      success: true,
      document,
    });
  } catch (error) {
    console.error(
      "Builder document GET error:",
      error,
    );

    return jsonError(
      "Unable to load builder document.",
      500,
    );
  }
}

export async function POST(request: NextRequest) {
  const originError = verifyRequestOrigin(request);

  if (originError) return originError;

  const auth = await authorize();

  if (auth.response) return auth.response;

  let body: SaveBody;

  try {
    body = (await request.json()) as SaveBody;
  } catch {
    return jsonError("Invalid JSON request body.");
  }

  if (!isValidDocumentKey(body.documentKey)) {
    return jsonError(
      "A valid documentKey is required.",
    );
  }

  // Restore a saved revision.
  if (body.action === "restore") {
    if (
      typeof body.revisionId !== "string" ||
      !body.revisionId.trim()
    ) {
      return jsonError(
        "A valid revisionId is required.",
      );
    }

    try {
      const restored = await prisma.$transaction(
        async (tx) => {
          const existing =
            await tx.builderDocument.findUnique({
              where: {
                documentKey: body.documentKey!,
              },
            });

          if (!existing) {
            throw new Error(
              "Builder document was not found.",
            );
          }

          const revision =
            await tx.builderRevision.findFirst({
              where: {
                id: body.revisionId!,
                documentId: existing.id,
              },
            });

          if (!revision) {
            throw new Error(
              "Selected revision was not found.",
            );
          }

          const snapshot =
            revision.snapshot as unknown as Snapshot;

          if (!snapshot || !isValidLayout(snapshot.layout)) {
            throw new Error(
              "The selected revision has an invalid layout.",
            );
          }

          const layout =
            snapshot.layout as Prisma.InputJsonValue;

          const settings =
            (snapshot.settings ?? {}) as Prisma.InputJsonValue;

          const globalStyles =
            (snapshot.globalStyles ?? {}) as Prisma.InputJsonValue;

          const conditions =
            (snapshot.conditions ?? {
              include: [],
              exclude: [],
            }) as Prisma.InputJsonValue;

          const restoredDocument =
            await tx.builderDocument.update({
              where: {
                id: existing.id,
              },
              data: {
                layout,
                settings,
                globalStyles,
                conditions,
                status:
                  snapshot.status === "PUBLISHED"
                    ? "PUBLISHED"
                    : "DRAFT",
                title: revision.title || existing.title,
              },
            });

          const latest =
            await tx.builderRevision.aggregate({
              where: {
                documentId: existing.id,
              },
              _max: {
                revisionNo: true,
              },
            });

          const revisionNo =
            (latest._max.revisionNo ?? 0) + 1;

          await tx.builderRevision.create({
            data: {
              documentId: existing.id,
              revisionNo,
              title: restoredDocument.title,
              snapshot: {
                layout,
                settings,
                globalStyles,
                conditions,
                status: restoredDocument.status,
              } as Prisma.InputJsonValue,
              message: `Restored revision ${revision.revisionNo}`,
              createdById: auth.user!.id,
            },
          });

          return restoredDocument;
        },
      );

      return NextResponse.json({
        success: true,
        message: "Revision restored successfully.",
        document: restored,
      });
    } catch (error) {
      console.error(
        "Builder revision restore error:",
        error,
      );

      return jsonError(
        error instanceof Error
          ? error.message
          : "Unable to restore revision.",
        500,
      );
    }
  }

  // Normal save or publish.
  if (
    !body.kind ||
    !ALLOWED_KINDS.includes(body.kind)
  ) {
    return jsonError(
      "Invalid builder document kind.",
    );
  }

  if (
    typeof body.title !== "string" ||
    !body.title.trim() ||
    body.title.length > 255
  ) {
    return jsonError(
      "Document title is required and must be under 255 characters.",
    );
  }

  if (
    body.status !== "DRAFT" &&
    body.status !== "PUBLISHED"
  ) {
    return jsonError(
      "Invalid document status.",
    );
  }

  try {
    if (!isValidLayout(body.layout)) {
      return jsonError("Invalid builder layout.");
    }
  } catch (error) {
    return jsonError(
      error instanceof Error
        ? error.message
        : "Invalid builder layout.",
    );
  }

  const layout =
    body.layout as Prisma.InputJsonValue;

  const settings =
    (body.settings ?? {}) as Prisma.InputJsonValue;

  const globalStyles =
    (body.globalStyles ?? {}) as Prisma.InputJsonValue;

  const conditions =
    (body.conditions ?? {
      include: [],
      exclude: [],
    }) as Prisma.InputJsonValue;

  try {
    const savedDocument =
      await prisma.$transaction(async (tx) => {
        const existing =
          await tx.builderDocument.findUnique({
            where: {
              documentKey: body.documentKey!,
            },
          });

        if (
          existing &&
          existing.kind !== body.kind
        ) {
          throw new Error(
            "Document key is already used by another builder type.",
          );
        }

        const document =
          await tx.builderDocument.upsert({
            where: {
              documentKey: body.documentKey!,
            },
            create: {
              documentKey: body.documentKey!,
              kind: body.kind!,
              title: body.title!.trim(),
              slug: body.slug ?? null,
              targetId: body.targetId ?? null,
              postTypeSlug: body.postTypeSlug ?? null,
              status: body.status!,
              layout,
              settings,
              globalStyles,
              conditions,
              publishedAt:
                body.status === "PUBLISHED"
                  ? new Date()
                  : null,
              createdById: auth.user!.id,
            },
            update: {
              title: body.title!.trim(),
              slug: body.slug ?? null,
              targetId: body.targetId ?? null,
              postTypeSlug: body.postTypeSlug ?? null,
              status: body.status!,
              layout,
              settings,
              globalStyles,
              conditions,
              ...(body.status === "PUBLISHED"
                ? { publishedAt: new Date() }
                : {}),
            },
          });

        if (body.status === "PUBLISHED") {
          const latest =
            await tx.builderRevision.aggregate({
              where: {
                documentId: document.id,
              },
              _max: {
                revisionNo: true,
              },
            });

          const revisionNo =
            (latest._max.revisionNo ?? 0) + 1;

          await tx.builderRevision.create({
            data: {
              documentId: document.id,
              revisionNo,
              title: document.title,
              snapshot: {
                layout,
                settings,
                globalStyles,
                conditions,
                status: body.status,
              } as Prisma.InputJsonValue,
              message:
                body.revisionMessage?.trim() ||
                `Published revision ${revisionNo}`,
              createdById: auth.user!.id,
            },
          });
        }

        return document;
      });

    return NextResponse.json({
      success: true,
      message:
        body.status === "PUBLISHED"
          ? "Builder document published successfully."
          : "Builder draft saved successfully.",
      document: savedDocument,
    });
  } catch (error) {
    console.error(
      "Builder document POST error:",
      error,
    );

    return jsonError(
      error instanceof Error
        ? error.message
        : "Unable to save builder document.",
      500,
    );
  }
}
